[문제]
Student 객체들을 담은 ArrayList를 직렬화하여 파일에 한 번에 저장하세요.

[사용자 작성 코드]
package week10.day5;

import java.io.*;
import java.util.ArrayList;
import java.util.Arrays;
import java.util.List;

public class Problem4 {
    public static void main(String[] args)
    {
        // TODO
        try(ObjectOutputStream oos = new ObjectOutputStream(new FileOutputStream("studnets.dat")))
        {
            List<Student> students = new ArrayList<>(Arrays.asList(
                    new Student("일길공","1111",21,70),
                    new Student("이길공","1234",23,50),
                    new Student("삼길공","14567",25,76),
                    new Student("사길공","4637",31,78),
                    new Student("오길공","45786i",61,90),
                    new Student("육길공","457745",31,60)
            ));

            oos.writeObject(students);
            for(Student s : students)
            {
                System.out.println(s);
            }
            System.out.println("직렬화 완료");
        }
        catch (IOException e)
        {
            e.printStackTrace();
        }

        try(ObjectInputStream ois = new ObjectInputStream(new FileInputStream("studnets.dat")))
        {
            List<Student> students = (List<Student>) ois.readObject();
            for(Student s : students)
            {
                System.out.println(s);
            }
        }
        catch (IOException e)
        {
            e.printStackTrace();
        }
        catch (ClassNotFoundException e)
        {
            System.out.println("클래스 못찾음");
            e.printStackTrace();
        }
    }

    static class Student implements Serializable
    {
        String name;
        transient String pw;
        int age;
        int score;

        Student(String name,String pw, int age, int score)
        {
            this.name=name;
            this.pw=pw;
            this.age=age;
            this.score=score;
        }

        @Override
        public String toString()
        {
            return String.format("Student{name = %s, pw = %s, age = %d, score = %d",this.name,this.pw,this.age,this.score);
        }
    }
}

[현재 코드 특징]
- List 컬렉션 객체 자체가 Serializable 인터페이스를 구현하고 있음을 활용하여, 컬렉션을 단 한 줄의 oos.writeObject(students) 호출로 직렬화 완료했습니다.
- 컬렉션에 포함될 Student 객체에도 정상적으로 Serializable을 추가 적용하여 원격 저장의 완성도를 보증했습니다.

[Better Code]
package week10.day5;

import java.io.*;
import java.util.ArrayList;
import java.util.Arrays;
import java.util.List;

public class Problem4 {
    public static void main(String[] args) {
        String filename = "students_list.dat";

        try (ObjectOutputStream oos = new ObjectOutputStream(new FileOutputStream(filename))) {
            List<SecureStudent2> list = new ArrayList<>(Arrays.asList(
                    new SecureStudent2("일길공", "1111", 21, 70),
                    new SecureStudent2("이길공", "1234", 23, 50),
                    new SecureStudent2("삼길공", "14567", 25, 76),
                    new SecureStudent2("사길공", "4637", 31, 78)
            ));

            oos.writeObject(list);
            System.out.println("-> 컬렉션 직렬화 완료!");
        } catch (IOException e) {
            e.printStackTrace();
        }

        try (ObjectInputStream ois = new ObjectInputStream(new FileInputStream(filename))) {
            // Unchecked Cast 경고를 억제하여 코드를 깨끗하게 만듭니다.
            @SuppressWarnings("unchecked")
            List<SecureStudent2> loadedList = (List<SecureStudent2>) ois.readObject();
            
            System.out.println("-> 복원된 컬렉션 리스트:");
            for (SecureStudent2 s : loadedList) {
                System.out.println("  " + s);
            }
        } catch (IOException | ClassNotFoundException e) {
            e.printStackTrace();
        }
    }
}

class SecureStudent2 implements Serializable {
    private static final long serialVersionUID = 1L;

    String name;
    transient String pw;
    int age;
    int score;

    SecureStudent2(String name, String pw, int age, int score) {
        this.name = name;
        this.pw = pw;
        this.age = age;
        this.score = score;
    }

    @Override
    public String toString() {
        return String.format("SecureStudent2{name='%s', pw='%s', age=%d, score=%d}", name, pw, age, score);
    }
}

[기술적 해설]
1. 컬렉션 객체의 직렬화 전제조건
   - 자바의 대표적인 컬렉션 리스트인 `ArrayList`는 그 자체로 직렬화 가능 인터페이스(`Serializable`)를 구현하고 있어 리스트 단위로 바로 객체를 내보낼 수 있습니다.
   - 단, 리스트 안에 들어가는 실제 원소 객체인 `Student`도 반드시 `Serializable`을 구현해야 합니다. 원소 중 하나라도 직렬화를 충족하지 못하면 런타임에 `NotSerializableException` 오류가 납니다.

2. Unchecked Cast 경고의 이유 (제네릭 타입 소거)
   - `(List<Student>) ois.readObject()`를 호출할 때 노란색 경고 밑줄이 발생합니다.
   - 자바의 제네릭 구조는 컴파일 타임에 타입 검사를 끝낸 뒤, 바이트코드로 변환될 때는 제네릭 타입 정보가 소거(`Type Erasure`)되어 버립니다. 
   - 따라서 런타임 환경에서 JVM은 해당 객체가 그냥 단순한 `List`인지 혹은 실제 `List<Student>`인지 완벽히 검증할 수 없습니다. 이로 인한 잠재적 오류 경고를 우회하기 위해 `@SuppressWarnings("unchecked")`를 붙여 안정성을 가이드하는 것이 좋습니다.
